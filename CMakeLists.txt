cmake_minimum_required(VERSION 3.20)
project(DredmorMac VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(EXPAT REQUIRED)
add_library(dredmor_core src/world.cpp src/random.cpp src/xml.cpp src/catalog.cpp src/stats.cpp)
target_link_libraries(dredmor_core PUBLIC EXPAT::EXPAT)
target_include_directories(dredmor_core PUBLIC include)
target_compile_options(dredmor_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(dredmor-replay src/replay.cpp)
target_link_libraries(dredmor-replay PRIVATE dredmor_core)
add_executable(dredmor-distance src/distance_cli.cpp)
target_link_libraries(dredmor-distance PRIVATE dredmor_core)
add_executable(dredmor-rng src/random_cli.cpp)
target_link_libraries(dredmor-rng PRIVATE dredmor_core)
add_executable(dredmor-data src/data_cli.cpp)
target_link_libraries(dredmor-data PRIVATE dredmor_core)
add_executable(dredmor-stats src/stats_cli.cpp)
target_link_libraries(dredmor-stats PRIVATE dredmor_core)
add_executable(dredmor-character src/character_cli.cpp)
target_link_libraries(dredmor-character PRIVATE dredmor_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(core-tests tests/core.cpp)
  target_link_libraries(core-tests PRIVATE dredmor_core)
  add_test(NAME core COMMAND core-tests)
  add_executable(data-tests tests/data.cpp)
  target_link_libraries(data-tests PRIVATE dredmor_core)
  add_test(NAME databases COMMAND data-tests ${CMAKE_SOURCE_DIR}/tests/fixtures/data-root)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME python-tools COMMAND ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_SOURCE_DIR}/tests -p "test_*.py")
endif()
if(APPLE)
  enable_language(OBJCXX)
  set(CMAKE_OBJCXX_STANDARD 17)
  set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
  add_executable(DredmorMac MACOSX_BUNDLE platform/macos/main.mm)
  target_link_libraries(DredmorMac PRIVATE dredmor_core "-framework Cocoa")
  target_compile_options(DredmorMac PRIVATE -fobjc-arc)
  set_target_properties(DredmorMac PROPERTIES
    MACOSX_BUNDLE_GUI_IDENTIFIER org.dredmormac.engine
    MACOSX_BUNDLE_BUNDLE_NAME DredmorMac)
  if(BUILD_TESTING)
    add_test(NAME native-fixture COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_SOURCE_DIR}/scripts/make-smoke-scene.py ${CMAKE_BINARY_DIR}/native-smoke)
    set_tests_properties(native-fixture PROPERTIES FIXTURES_SETUP native-scene)
    add_test(NAME native-window COMMAND $<TARGET_FILE:DredmorMac>
      ${CMAKE_BINARY_DIR}/native-smoke/scene.json --smoke-test)
    set_tests_properties(native-window PROPERTIES FIXTURES_REQUIRED native-scene TIMEOUT 15)
  endif()
endif()
