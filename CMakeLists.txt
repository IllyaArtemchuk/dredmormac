cmake_minimum_required(VERSION 3.20)
project(DredmorMac VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(dredmor_core src/world.cpp)
target_include_directories(dredmor_core PUBLIC include)
target_compile_options(dredmor_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(dredmor-replay src/replay.cpp)
target_link_libraries(dredmor-replay PRIVATE dredmor_core)
add_executable(dredmor-distance src/distance_cli.cpp)
target_link_libraries(dredmor-distance PRIVATE dredmor_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(core-tests tests/core.cpp)
  target_link_libraries(core-tests PRIVATE dredmor_core)
  add_test(NAME core COMMAND core-tests)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME python-tools COMMAND ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_SOURCE_DIR}/tests -p "test_*.py")
endif()
if(APPLE)
  enable_language(OBJCXX)
  set(CMAKE_OBJCXX_STANDARD 17)
  set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
  add_executable(DredmorMac MACOSX_BUNDLE platform/macos/main.mm)
  target_link_libraries(DredmorMac PRIVATE dredmor_core "-framework Cocoa")
  target_compile_options(DredmorMac PRIVATE -fobjc-arc)
  set_target_properties(DredmorMac PROPERTIES
    MACOSX_BUNDLE_GUI_IDENTIFIER org.dredmormac.engine
    MACOSX_BUNDLE_BUNDLE_NAME DredmorMac)
endif()
